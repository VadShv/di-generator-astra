   178→
   179→  // Wizard state
   180→  const [wizardStep, setWizardStep] = useState<WizardStep>('position')
   181→  const [selectedPositionId, setSelectedPositionId] = useState('')
   182→  const [selectedTemplateId, setSelectedTemplateId] = useState('')
   183→  const [resolvedPrompt, setResolvedPrompt] = useState<MasterPrompt | null>(null)
   184→  const [overridePromptId, setOverridePromptId] = useState<string>('')
   185→  const [generationMode, setGenerationMode] = useState<'all' | 'section'>('all')
   186→  const [generating, setGenerating] = useState(false)
   187→  const [generationProgress, setGenerationProgress] = useState(0)
   188→  const [sectionGenerating, setSectionGenerating] = useState<Record<number, boolean>>({})
   189→  const [sectionCustomPrompts, setSectionCustomPrompts] = useState<Record<number, string>>({})
   190→
   191→  // Editor state
   192→  const [editingDI, setEditingDI] = useState<GeneratedDI | null>(null)
   193→  const [editSections, setEditSections] = useState<GeneratedDISection[]>([])
   194→  const [editTitle, setEditTitle] = useState('')
   195→  const [improvingSectionId, setImprovingSectionId] = useState<string | null>(null)
   196→  const [improveInstruction, setImproveInstruction] = useState('')
   197→  const [improveDialogOpen, setImproveDialogOpen] = useState(false)
   198→  const [activeImproveSectionId, setActiveImproveSectionId] = useState<string>('')
   199→
   200→  // Delete confirmation
   201→  const [deleteId, setDeleteId] = useState<string | null>(null)
   202→
   203→  // Search
   204→  const [searchQuery, setSearchQuery] = useState('')
   205→
   206→  // Data fetching
   207→  const fetchGeneratedDIs = useCallback(async () => {
   208→    try {
   209→      const res = await fetch('/api/generate-di')
   210→      if (res.ok) {
   211→        const data = await res.json()
   212→        setGeneratedDIs(data)
   213→      }
   214→    } catch {
   215→      toast({ title: 'Ошибка', description: 'Не удалось загрузить сгенерированные ДИ', variant: 'destructive' })
   216→    }
   217→  }, [toast])
   218→
   219→  const fetchPositions = useCallback(async () => {
   220→    try {
   221→      const res = await fetch('/api/positions')
   222→      if (res.ok) {
   223→        const data = await res.json()
   224→        setPositions(data)
   225→      }
   226→    } catch {
   227→      toast({ title: 'Ошибка', description: 'Не удалось загрузить должности', variant: 'destructive' })
   228→    }
   229→  }, [toast])
   230→
   231→  const fetchTemplates = useCallback(async () => {
   232→    try {
   233→      const res = await fetch('/api/templates')
   234→      if (res.ok) {
   235→        const data = await res.json()
   236→        setTemplates(data.filter((t: Template) => t.isActive))
   237→      }
   238→    } catch {
   239→      toast({ title: 'Ошибка', description: 'Не удалось загрузить шаблоны', variant: 'destructive' })
   240→    }
   241→  }, [toast])
   242→
   243→  const fetchMasterPrompts = useCallback(async () => {
   244→    try {
   245→      const res = await fetch('/api/master-prompts?active=true')
   246→      if (res.ok) {
   247→        const data = await res.json()
   248→        setMasterPrompts(Array.isArray(data) ? data : [])
   249→      }
   250→    } catch {
   251→      // Non-critical, just set empty
   252→      setMasterPrompts([])
   253→    }
   254→  }, [])
   255→
   256→  const resolvePrompt = useCallback(async (positionId: string) => {
   257→    const position = positions.find((p) => p.id === positionId)
   258→    if (!position) return
   259→    try {
   260→      const params = new URLSearchParams()
   261→      params.set('departmentId', position.departmentId)
   262→      if (position.domain) params.set('domain', position.domain)
   263→      if (position.grade) params.set('grade', position.grade)
   264→      const res = await fetch(`/api/master-prompts?${params.toString()}`)
   265→      if (res.ok) {
   266→        const data = await res.json()
   267→        setResolvedPrompt(data)
   268→        setOverridePromptId('')
   269→      }
   270→    } catch {
   271→      // Non-critical
   272→    }
   273→  }, [positions])
   274→
   275→  useEffect(() => {
   276→    const loadData = async () => {
   277→      setLoading(true)
   278→      await Promise.all([fetchGeneratedDIs(), fetchPositions(), fetchTemplates(), fetchMasterPrompts()])
   279→      setLoading(false)
   280→    }
   281→    loadData()
   282→  }, [fetchGeneratedDIs, fetchPositions, fetchTemplates, fetchMasterPrompts])
   283→
   284→  // Resolve prompt when position changes in wizard
   285→  useEffect(() => {
   286→    if (selectedPositionId) {
   287→      resolvePrompt(selectedPositionId)
   288→    }
   289→  }, [selectedPositionId, resolvePrompt])
   290→
   291→  // ============================================
   292→  // Handlers
   293→  // ============================================
   294→
   295→  const startWizard = () => {

   297→    setSelectedPositionId('')
   298→    setSelectedTemplateId('')
   299→    setResolvedPrompt(null)
   300→    setOverridePromptId('')

   303→    setGenerationProgress(0)

   305→    setSectionCustomPrompts({})
   306→    setViewMode('wizard')
   307→  }
   308→
   309→  const handleGenerateAll = async () => {
   310→    setGenerating(true)
   311→    setGenerationProgress(10)
   312→    setWizardStep('generating')
   313→
   314→    try {
   315→      const res = await fetch('/api/generate-di/ai-generate', {
   316→        method: 'POST',
   317→        headers: { 'Content-Type': 'application/json' },
   318→        body: JSON.stringify({
   319→          positionId: selectedPositionId,
   320→          templateId: selectedTemplateId,
   321→        }),
   322→      })
   323→
   324→      setGenerationProgress(90)
   325→
   326→      if (!res.ok) {
   327→        const err = await res.json()
   328→        throw new Error(err.error || 'Ошибка генерации')
   329→      }
   330→
   331→      const data = await res.json()
   332→      setGenerationProgress(100)
   333→      setEditingDI(data)
   334→      setEditSections(data.sections || [])
   335→      setEditTitle(data.title)
   336→
   337→      toast({ title: 'Успех', description: 'ДИ успешно сгенерирована' })
   338→      setViewMode('editor')
   339→    } catch (error) {
   340→      toast({
   341→        title: 'Ошибка генерации',
   342→        description: error instanceof Error ? error.message : 'Не удалось сгенерировать ДИ',
   343→        variant: 'destructive',
   344→      })
   345→      setWizardStep('mode')
   346→    } finally {
   347→      setGenerating(false)
   348→    }
   349→  }
   350→
   351→  const handleGenerateSection = async (sectionOrder: number) => {
   352→    setSectionGenerating((prev) => ({ ...prev, [sectionOrder]: true }))
   353→
   354→    try {
   355→      // First create a draft DI if not yet created
   356→      let diId = editingDI?.id
   357→      if (!diId) {
   358→        const position = positions.find((p) => p.id === selectedPositionId)
   359→        const template = templates.find((t) => t.id === selectedTemplateId)
   360→        const sections = template?.sections.map((s) => ({
   361→          sectionTitle: s.title,
   362→          sectionContent: '',
   363→          order: s.order,
   364→        })) || []
   365→
   366→        const createRes = await fetch('/api/generate-di', {
   367→          method: 'POST',
   368→          headers: { 'Content-Type': 'application/json' },
   369→          body: JSON.stringify({
   370→            positionId: selectedPositionId,
   371→            templateId: selectedTemplateId,
   372→            title: `ДИ — ${position?.title || 'Должность'}`,
   373→            sections,
   374→          }),
   375→        })
   376→
   377→        if (!createRes.ok) throw new Error('Не удалось создать ДИ')
   378→        const created = await createRes.json()
   379→        diId = created.id
   380→        setEditingDI(created)
   381→        setEditSections(created.sections || [])
   382→        setEditTitle(created.title)
   383→      }
   384→
   385→      const res = await fetch('/api/generate-di/ai-section', {
   386→        method: 'POST',
   387→        headers: { 'Content-Type': 'application/json' },
   388→        body: JSON.stringify({
   389→          generatedDIId: diId,
   390→          sectionOrder,
   391→          customPrompt: sectionCustomPrompts[sectionOrder] || undefined,
   392→        }),
   393→      })
   394→
   395→      if (!res.ok) {
   396→        const err = await res.json()
   397→        throw new Error(err.error || 'Ошибка генерации секции')
   398→      }
   399→
   400→      const updatedSection = await res.json()
   401→      setEditSections((prev) =>
   402→        prev.map((s) => (s.order === sectionOrder ? { ...s, ...updatedSection } : s))
   403→      )
   404→
   405→      toast({ title: 'Секция сгенерирована', description: updatedSection.sectionTitle })
   406→    } catch (error) {
   407→      toast({
   408→        title: 'Ошибка',
   409→        description: error instanceof Error ? error.message : 'Не удалось сгенерировать секцию',
   410→        variant: 'destructive',
   411→      })
   412→    } finally {
   413→      setSectionGenerating((prev) => ({ ...prev, [sectionOrder]: false }))
   414→    }
   415→  }
   416→
   417→  const handleImproveSection = async (sectionId: string, instruction: string) => {

   420→      const res = await fetch('/api/generate-di/ai-improve', {

   422→        headers: { 'Content-Type': 'application/json' },

   424→      })

   426→      if (!res.ok) {
   427→        const err = await res.json()
   428→        throw new Error(err.error || 'Ошибка улучшения')
   429→      }
   430→
   431→      const updatedSection = await res.json()
   432→      setEditSections((prev) =>
   433→        prev.map((s) => (s.id === sectionId ? { ...s, ...updatedSection } : s))

   435→
   436→      toast({ title: 'Секция улучшена', description: 'Текст секции обновлён' })

   438→      setImproveInstruction('')
   439→    } catch (error) {
   440→      toast({
   441→        title: 'Ошибка',
   442→        description: error instanceof Error ? error.message : 'Не удалось улучшить секцию',
   443→        variant: 'destructive',
   444→      })
   445→    } finally {
   446→      setImprovingSectionId(null)
   447→    }
   448→  }

   454→      const res = await fetch('/api/generate-di/ai-section', {
   455→        method: 'POST',
   456→        headers: { 'Content-Type': 'application/json' },
   457→        body: JSON.stringify({
   458→          generatedDIId: editingDI.id,
   459→          sectionOrder: section.order,
   460→        }),
   461→      })
   462→
   463→      if (!res.ok) {
   464→        const err = await res.json()
   465→        throw new Error(err.error || 'Ошибка перегенерации')
   466→      }
   467→
   468→      const updatedSection = await res.json()
   469→      setEditSections((prev) =>
   470→        prev.map((s) => (s.id === section.id ? { ...s, ...updatedSection } : s))
   471→      )
   472→
   473→      toast({ title: 'Секция перегенерирована', description: section.sectionTitle })
   474→    } catch (error) {
   475→      toast({
   476→        title: 'Ошибка',
   477→        description: error instanceof Error ? error.message : 'Не удалось перегенерировать секцию',
   478→        variant: 'destructive',
   479→      })
   480→    } finally {
   481→      setSectionGenerating((prev) => ({ ...prev, [section.order]: false }))
   482→    }
   483→  }
   484→
   485→  const handleSaveDI = async () => {

   487→    try {
   488→      const res = await fetch('/api/generate-di', {
   489→        method: 'PUT',
   490→        headers: { 'Content-Type': 'application/json' },

   492→          id: editingDI.id,
   493→          title: editTitle,
   494→          sections: editSections.map((s) => ({
   495→            sectionTitle: s.sectionTitle,
   496→            sectionContent: s.sectionContent,
   497→            order: s.order,
   498→            aiGenerated: s.aiGenerated,
   499→            editedBy: s.editedBy,
   500→          })),
   501→        }),
   502→      })
   503→
   504→      if (!res.ok) throw new Error('Ошибка сохранения')
   505→
   506→      toast({ title: 'Сохранено', description: 'ДИ успешно сохранена' })
   507→      fetchGeneratedDIs()
   508→    } catch {
   509→      toast({ title: 'Ошибка', description: 'Не удалось сохранить ДИ', variant: 'destructive' })
   510→    }
   511→  }
   512→
   513→  const handleStatusChange = async (newStatus: string) => {
   514→    if (!editingDI) return
   515→    try {
   516→      const res = await fetch('/api/generate-di', {
   517→        method: 'PUT',
   518→        headers: { 'Content-Type': 'application/json' },
   519→        body: JSON.stringify({ id: editingDI.id, status: newStatus }),
   520→      })
   521→
   522→      if (!res.ok) throw new Error('Ошибка изменения статуса')
   523→
   524→      const updated = await res.json()
   525→      setEditingDI(updated)
   526→      toast({ title: 'Статус обновлён', description: STATUS_CONFIG[newStatus]?.label || newStatus })
   527→      fetchGeneratedDIs()
   528→    } catch {
   529→      toast({ title: 'Ошибка', description: 'Не удалось изменить статус', variant: 'destructive' })
   530→    }
   531→  }

   535→      const res = await fetch('/api/generate-di', {
   536→        method: 'DELETE',
   537→        headers: { 'Content-Type': 'application/json' },
   538→        body: JSON.stringify({ id }),
   539→      })
   540→
   541→      if (!res.ok) throw new Error('Ошибка удаления')
   542→
   543→      toast({ title: 'Удалено', description: 'ДИ успешно удалена' })
   544→      fetchGeneratedDIs()
   545→    } catch {
   546→      toast({ title: 'Ошибка', description: 'Не удалось удалить ДИ', variant: 'destructive' })
   547→    } finally {
   548→      setDeleteId(null)
   549→    }
   550→  }

   552→  const openEditor = (di: GeneratedDI) => {

   554→    setEditSections([...di.sections])
   555→    setEditTitle(di.title)
   556→    setViewMode('editor')

   564→    { key: 'position', label: 'Должность', icon: <FileText className="h-4 w-4" /> },
   565→    { key: 'template', label: 'Шаблон', icon: <BookOpen className="h-4 w-4" /> },
   566→    { key: 'prompt', label: 'Промпт', icon: <Brain className="h-4 w-4" /> },
   567→    { key: 'mode', label: 'Режим', icon: <Wand2 className="h-4 w-4" /> },
   568→    { key: 'generating', label: 'Генерация', icon: <Sparkles className="h-4 w-4" /> },
   569→    { key: 'review', label: 'Результат', icon: <Check className="h-4 w-4" /> },
   570→  ]
   571→
   572→  const stepIndex = WIZARD_STEPS.findIndex((s) => s.key === wizardStep)
   573→
   574→  const renderStepIndicator = () => (
