     1→'use client'
     2→
     3→import { useEffect, useState, useCallback } from 'react'
     4→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Button } from '@/components/ui/button'
     6→import { Input } from '@/components/ui/input'
     7→import { Label } from '@/components/ui/label'
     8→import { Textarea } from '@/components/ui/textarea'
     9→import { Badge } from '@/components/ui/badge'
    10→import {
    11→  Table,
    12→  TableBody,
    13→  TableCell,
    14→  TableHead,
    15→  TableHeader,
    16→  TableRow,
    17→} from '@/components/ui/table'
    18→import {
    19→  Dialog,
    20→  DialogContent,
    21→  DialogDescription,
    22→  DialogFooter,
    23→  DialogHeader,
    24→  DialogTitle,
    25→} from '@/components/ui/dialog'
    26→import {
    27→  AlertDialog,
    28→  AlertDialogAction,
    29→  AlertDialogCancel,
    30→  AlertDialogContent,
    31→  AlertDialogDescription,
    32→  AlertDialogFooter,
    33→  AlertDialogHeader,
    34→  AlertDialogTitle,
    35→} from '@/components/ui/alert-dialog'
    36→import {
    37→  Select,
    38→  SelectContent,
    39→  SelectItem,
    40→  SelectTrigger,
    41→  SelectValue,
    42→} from '@/components/ui/select'
    43→import { ScrollArea } from '@/components/ui/scroll-area'
    44→import { Separator } from '@/components/ui/separator'
    45→import { useToast } from '@/hooks/use-toast'
    46→import {
    47→  Archive,
    48→  Plus,
    49→  Search,
    50→  Eye,
    51→  Pencil,
    52→  Trash2,
    53→  FileText,
    54→  Upload,
    55→  X,
    56→  FolderOpen,
    57→  AlertCircle,
    58→  CheckCircle2,
    59→} from 'lucide-react'
    60→
    61→// ============================================
    62→// Types
    63→// ============================================
    64→
    65→interface Department {
    66→  id: string
    67→  name: string
    68→  code: string
    69→}
    70→
    71→interface Position {
    72→  id: string
    73→  title: string
    74→  code: string
    75→  departmentId: string
    76→  department: Department
    77→  grade: string | null
    78→  domain: string | null
    79→}
    80→
    81→interface ArchiveDI {
    82→  id: string
    83→  title: string
    84→  content: string
    85→  positionId: string
    86→  fileName: string | null
    87→  uploadedAt: string
    88→  createdAt: string
    89→  updatedAt: string
    90→  position: Position
    91→}
    92→
    93→interface Stats {
    94→  totalArchiveDIs: number
    95→  positionsWithDIs: number
    96→  positionsWithoutDIs: number
    97→}
    98→
    99→// ============================================
   100→// Component
   101→// ============================================
   102→
   103→export function ArchiveModule() {
   104→  const { toast } = useToast()
   105→
   106→  // Data state
   107→  const [archiveDIs, setArchiveDIs] = useState<ArchiveDI[]>([])
   108→  const [positions, setPositions] = useState<Position[]>([])
   109→  const [departments, setDepartments] = useState<Department[]>([])
   110→  const [stats, setStats] = useState<Stats>({ totalArchiveDIs: 0, positionsWithDIs: 0, positionsWithoutDIs: 0 })
   111→  const [loading, setLoading] = useState(true)
   112→
   113→  // Filter state
   114→  const [searchText, setSearchText] = useState('')
   115→  const [filterPositionId, setFilterPositionId] = useState<string>('all')
   116→  const [filterDepartmentId, setFilterDepartmentId] = useState<string>('all')
   117→
   118→  // Dialog state
   119→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
   120→  const [viewDialogOpen, setViewDialogOpen] = useState(false)
   121→  const [editDialogOpen, setEditDialogOpen] = useState(false)
   122→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   123→
   124→  // Form state for upload
   125→  const [uploadTitle, setUploadTitle] = useState('')
   126→  const [uploadContent, setUploadContent] = useState('')
   127→  const [uploadPositionId, setUploadPositionId] = useState('')
   128→  const [uploadFileName, setUploadFileName] = useState('')
   129→
   130→  // Form state for edit
   131→  const [editId, setEditId] = useState('')
   132→  const [editTitle, setEditTitle] = useState('')
   133→  const [editContent, setEditContent] = useState('')
   134→  const [editPositionId, setEditPositionId] = useState('')
   135→
   136→  // View state
   137→  const [viewDI, setViewDI] = useState<ArchiveDI | null>(null)
   138→
   139→  // Delete state
   140→  const [deleteId, setDeleteId] = useState('')
   141→  const [deleteTitle, setDeleteTitle] = useState('')
   142→
   143→  // Saving states
   144→  const [uploading, setUploading] = useState(false)
   145→  const [editing, setEditing] = useState(false)
   146→
   147→  // ============================================
   148→  // Data fetching
   149→  // ============================================
   150→
   151→  const fetchArchiveDIs = useCallback(async () => {
   152→    try {
   153→      const params = new URLSearchParams()
   154→      if (filterPositionId && filterPositionId !== 'all') params.set('positionId', filterPositionId)
   155→      if (searchText) params.set('search', searchText)
   156→
   157→      const res = await fetch(`/api/archive-di?${params.toString()}`)
   158→      if (res.ok) {
   159→        const data = await res.json()
   160→        setArchiveDIs(data)
   161→      }
   162→    } catch (error) {
   163→      console.error('Error fetching archive DIs:', error)
   164→    }
   165→  }, [filterPositionId, searchText])
   166→
   167→  const fetchPositions = useCallback(async () => {
   168→    try {
   169→      const params = new URLSearchParams()
   170→      if (filterDepartmentId && filterDepartmentId !== 'all') params.set('departmentId', filterDepartmentId)
   171→      const res = await fetch(`/api/positions?${params.toString()}`)
   172→      if (res.ok) {
   173→        const data = await res.json()
   174→        setPositions(data)
   175→      }
   176→    } catch (error) {
   177→      console.error('Error fetching positions:', error)
   178→    }
   179→  }, [filterDepartmentId])
   180→
   181→  const fetchDepartments = useCallback(async () => {
   182→    try {
   183→      const res = await fetch('/api/departments')
   184→      if (res.ok) {
   185→        const data = await res.json()
   186→        setDepartments(data)
   187→      }
   188→    } catch (error) {
   189→      console.error('Error fetching departments:', error)
   190→    }
   191→  }, [])
   192→
   193→  const fetchStats = useCallback(async () => {
   194→    try {
   195→      const [allDIsRes, allPositionsRes] = await Promise.all([
   196→        fetch('/api/archive-di'),
   197→        fetch('/api/positions'),
   198→      ])
   199→
   200→      if (allDIsRes.ok && allPositionsRes.ok) {
   201→        const allDIs: ArchiveDI[] = await allDIsRes.json()
   202→        const allPositions: Position[] = await allPositionsRes.json()
   203→
   204→        const positionsWithDIs = new Set(allDIs.map((di: ArchiveDI) => di.positionId)).size
   205→        const positionsWithoutDIs = allPositions.length - positionsWithDIs
   206→
   207→        setStats({
   208→          totalArchiveDIs: allDIs.length,
   209→          positionsWithDIs,
   210→          positionsWithoutDIs: Math.max(0, positionsWithoutDIs),
   211→        })
   212→      }
   213→    } catch (error) {
   214→      console.error('Error fetching stats:', error)
   215→    }
   216→  }, [])
   217→
   218→  // ============================================
   219→  // Effects
   220→  // ============================================
   221→
   222→  useEffect(() => {
   223→    async function initialLoad() {
   224→      setLoading(true)
   225→      await Promise.all([fetchDepartments(), fetchPositions()])
   226→      setLoading(false)
   227→    }
   228→    initialLoad()
   229→  }, [fetchDepartments, fetchPositions])
   230→
   231→  useEffect(() => {
   232→    fetchArchiveDIs()
   233→  }, [fetchArchiveDIs])
   234→
   235→  useEffect(() => {
   236→    fetchPositions()
   237→  }, [fetchPositions])
   238→
   239→  useEffect(() => {
   240→    if (!loading) {
   241→      fetchStats()
   242→    }
   243→  }, [loading, archiveDIs.length, fetchStats])
   244→
   245→  // ============================================
   246→  // Filtered data
   247→  // ============================================
   248→
   249→  const filteredDIs = archiveDIs.filter((di) => {
   250→    if (filterDepartmentId && filterDepartmentId !== 'all') {
   251→      if (di.position.departmentId !== filterDepartmentId) return false
   252→    }
   253→    return true
   254→  })
   255→
   256→  // ============================================
   257→  // Handlers
   258→  // ============================================
   259→
   260→  const handleUpload = async () => {
   261→    if (!uploadTitle.trim() || !uploadContent.trim() || !uploadPositionId) {
   262→      toast({
   263→        title: 'Ошибка',
   264→        description: 'Заполните все обязательные поля',
   265→        variant: 'destructive',
   266→      })
   267→      return
   268→    }
   269→
   270→    setUploading(true)
   271→    try {
   272→      const res = await fetch('/api/archive-di', {
   273→        method: 'POST',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({
   276→          title: uploadTitle.trim(),
   277→          content: uploadContent.trim(),
   278→          positionId: uploadPositionId,
   279→          fileName: uploadFileName.trim() || null,
   280→        }),
   281→      })
   282→
   283→      if (res.ok) {
   284→        toast({
   285→          title: 'Успешно',
   286→          description: 'Архивная ДИ добавлена',
   287→        })
   288→        setUploadDialogOpen(false)
   289→        resetUploadForm()
   290→        fetchArchiveDIs()
   291→        fetchStats()
   292→      } else {
   293→        const data = await res.json()
   294→        toast({
   295→          title: 'Ошибка',
   296→          description: data.error || 'Не удалось добавить ДИ',
   297→          variant: 'destructive',
   298→        })
   299→      }
   300→    } catch {
   301→      toast({
   302→        title: 'Ошибка',
   303→        description: 'Ошибка сети',
   304→        variant: 'destructive',
   305→      })
   306→    } finally {
   307→      setUploading(false)
   308→    }
   309→  }
   310→
   311→  const handleEdit = async () => {
   312→    if (!editTitle.trim() || !editContent.trim() || !editPositionId) {
   313→      toast({
   314→        title: 'Ошибка',
   315→        description: 'Заполните все обязательные поля',
   316→        variant: 'destructive',
   317→      })
   318→      return
   319→    }
   320→
   321→    setEditing(true)
   322→    try {
   323→      const res = await fetch('/api/archive-di', {
   324→        method: 'PUT',
   325→        headers: { 'Content-Type': 'application/json' },
   326→        body: JSON.stringify({
   327→          id: editId,
   328→          title: editTitle.trim(),
   329→          content: editContent.trim(),
   330→          positionId: editPositionId,
   331→        }),
   332→      })
   333→
   334→      if (res.ok) {
   335→        toast({
   336→          title: 'Успешно',
   337→          description: 'Архивная ДИ обновлена',
   338→        })
   339→        setEditDialogOpen(false)
   340→        fetchArchiveDIs()
   341→        fetchStats()
   342→      } else {
   343→        const data = await res.json()
   344→        toast({
   345→          title: 'Ошибка',
   346→          description: data.error || 'Не удалось обновить ДИ',
   347→          variant: 'destructive',
   348→        })
   349→      }
   350→    } catch {
   351→      toast({
   352→        title: 'Ошибка',
   353→        description: 'Ошибка сети',
   354→        variant: 'destructive',
   355→      })
   356→    } finally {
   357→      setEditing(false)
   358→    }
   359→  }
   360→
   361→  const handleDelete = async () => {
   362→    try {
   363→      const res = await fetch('/api/archive-di', {
   364→        method: 'DELETE',
   365→        headers: { 'Content-Type': 'application/json' },
   366→        body: JSON.stringify({ id: deleteId }),
   367→      })
   368→
   369→      if (res.ok) {
   370→        toast({
   371→          title: 'Успешно',
   372→          description: 'Архивная ДИ удалена',
   373→        })
   374→        setDeleteDialogOpen(false)
   375→        fetchArchiveDIs()
   376→        fetchStats()
   377→      } else {
   378→        const data = await res.json()
   379→        toast({
   380→          title: 'Ошибка',
   381→          description: data.error || 'Не удалось удалить ДИ',
   382→          variant: 'destructive',
   383→        })
   384→      }
   385→    } catch {
   386→      toast({
   387→        title: 'Ошибка',
   388→        description: 'Ошибка сети',
   389→        variant: 'destructive',
   390→      })
   391→    }
   392→  }
   393→
   394→  const handleView = async (di: ArchiveDI) => {
   395→    try {
   396→      const res = await fetch(`/api/archive-di/${di.id}`)
   397→      if (res.ok) {
   398→        const data = await res.json()
   399→        setViewDI(data)
   400→        setViewDialogOpen(true)
   401→      }
   402→    } catch {
   403→      // Fallback to the data we already have
   404→      setViewDI(di)
   405→      setViewDialogOpen(true)
   406→    }
   407→  }
   408→
   409→  const openEditDialog = (di: ArchiveDI) => {
   410→    setEditId(di.id)
   411→    setEditTitle(di.title)
   412→    setEditContent(di.content)
   413→    setEditPositionId(di.positionId)
   414→    setEditDialogOpen(true)
   415→  }
   416→
   417→  const openDeleteDialog = (di: ArchiveDI) => {
   418→    setDeleteId(di.id)
   419→    setDeleteTitle(di.title)
   420→    setDeleteDialogOpen(true)
   421→  }
   422→
   423→  const resetUploadForm = () => {
   424→    setUploadTitle('')
   425→    setUploadContent('')
   426→    setUploadPositionId('')
   427→    setUploadFileName('')
   428→  }
   429→
   430→  const formatDate = (dateStr: string) => {
   431→    try {
   432→      return new Date(dateStr).toLocaleDateString('ru-RU', {
   433→        day: '2-digit',
   434→        month: '2-digit',
   435→        year: 'numeric',
   436→      })
   437→    } catch {
   438→      return dateStr
   439→    }
   440→  }
   441→
   442→  // ============================================
   443→  // Render
   444→  // ============================================
   445→
   446→  return (
   447→    <div className="space-y-6">
   448→      {/* Header */}
   449→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   450→        <div>
   451→          <h1 className="text-3xl font-bold tracking-tight flex items-center gap-2">
   452→            <Archive className="h-8 w-8 text-amber-600" />
   453→            Архив ДИ
   454→          </h1>
   455→          <p className="text-muted-foreground mt-1">
   456→            Управление архивом должностных инструкций
   457→          </p>
   458→        </div>
   459→        <Button onClick={() => setUploadDialogOpen(true)} className="gap-2">
   460→          <Upload className="h-4 w-4" />
   461→          Добавить ДИ
   462→        </Button>
   463→      </div>
   464→
   465→      {/* Stats Cards */}
   466→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   467→        <Card className="hover:shadow-md transition-shadow">
   468→          <CardContent className="p-6">
   469→            <div className="flex items-center justify-between">
   470→              <div>
   471→                <p className="text-sm text-muted-foreground">Всего в архиве</p>
   472→                <p className="text-3xl font-bold mt-1">{stats.totalArchiveDIs}</p>
   473→              </div>
   474→              <div className="p-3 rounded-xl bg-amber-50 text-amber-600">
   475→                <Archive className="h-5 w-5" />
   476→              </div>
   477→            </div>
   478→          </CardContent>
   479→        </Card>
   480→        <Card className="hover:shadow-md transition-shadow">
   481→          <CardContent className="p-6">
   482→            <div className="flex items-center justify-between">
   483→              <div>
   484→                <p className="text-sm text-muted-foreground">Должности с ДИ</p>
   485→                <p className="text-3xl font-bold mt-1">{stats.positionsWithDIs}</p>
   486→              </div>
   487→              <div className="p-3 rounded-xl bg-emerald-50 text-emerald-600">
   488→                <CheckCircle2 className="h-5 w-5" />
   489→              </div>
   490→            </div>
   491→          </CardContent>
   492→        </Card>
   493→        <Card className="hover:shadow-md transition-shadow">
   494→          <CardContent className="p-6">
   495→            <div className="flex items-center justify-between">
   496→              <div>
   497→                <p className="text-sm text-muted-foreground">Без архивной ДИ</p>
   498→                <p className="text-3xl font-bold mt-1">{stats.positionsWithoutDIs}</p>
   499→              </div>
   500→              <div className="p-3 rounded-xl bg-rose-50 text-rose-600">
   501→                <AlertCircle className="h-5 w-5" />
   502→              </div>
   503→            </div>
   504→          </CardContent>
   505→        </Card>
   506→      </div>
   507→
   508→      {/* Filters */}
   509→      <Card>
   510→        <CardContent className="p-4">
   511→          <div className="flex flex-col sm:flex-row gap-3">
   512→            <div className="relative flex-1">
   513→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   514→              <Input
   515→                placeholder="Поиск по названию или содержимому..."
   516→                value={searchText}
   517→                onChange={(e) => setSearchText(e.target.value)}
   518→                className="pl-9"
   519→              />
   520→              {searchText && (
   521→                <Button
   522→                  variant="ghost"
   523→                  size="icon"
   524→                  className="absolute right-1 top-1/2 -translate-y-1/2 h-7 w-7"
   525→                  onClick={() => setSearchText('')}
   526→                >
   527→                  <X className="h-3 w-3" />
   528→                </Button>
   529→              )}
   530→            </div>
   531→            <Select value={filterDepartmentId} onValueChange={(val) => { setFilterDepartmentId(val); setFilterPositionId('all'); }}>
   532→              <SelectTrigger className="w-full sm:w-[220px]">
   533→                <SelectValue placeholder="Подразделение" />
   534→              </SelectTrigger>
   535→              <SelectContent>
   536→                <SelectItem value="all">Все подразделения</SelectItem>
   537→                {departments.map((dept) => (
   538→                  <SelectItem key={dept.id} value={dept.id}>
   539→                    {dept.name}
   540→                  </SelectItem>
   541→                ))}
   542→              </SelectContent>
   543→            </Select>
   544→            <Select value={filterPositionId} onValueChange={setFilterPositionId}>
   545→              <SelectTrigger className="w-full sm:w-[220px]">
   546→                <SelectValue placeholder="Должность" />
   547→              </SelectTrigger>
   548→              <SelectContent>
   549→                <SelectItem value="all">Все должности</SelectItem>
   550→                {positions.map((pos) => (
   551→                  <SelectItem key={pos.id} value={pos.id}>
   552→                    {pos.title}
   553→                  </SelectItem>
   554→                ))}
   555→              </SelectContent>
   556→            </Select>
   557→          </div>
   558→        </CardContent>
   559→      </Card>
   560→
   561→      {/* Table */}
   562→      <Card>
   563→        <CardContent className="p-0">
   564→          {loading ? (
   565→            <div className="p-8 text-center text-muted-foreground">
   566→              <div className="animate-pulse space-y-3">
   567→                <div className="h-10 bg-muted rounded" />
   568→                <div className="h-12 bg-muted rounded" />
   569→                <div className="h-12 bg-muted rounded" />
   570→                <div className="h-12 bg-muted rounded" />
   571→              </div>
   572→            </div>
   573→          ) : filteredDIs.length === 0 ? (
   574→            <div className="p-12 text-center">
   575→              <FolderOpen className="h-12 w-12 text-muted-foreground/50 mx-auto mb-3" />
   576→              <p className="text-muted-foreground text-lg">Нет архивных ДИ</p>
   577→              <p className="text-muted-foreground/70 text-sm mt-1">
   578→                Добавьте первую должностную инструкцию в архив
   579→              </p>
   580→              <Button
   581→                className="mt-4 gap-2"
   582→                onClick={() => setUploadDialogOpen(true)}
   583→              >
   584→                <Plus className="h-4 w-4" />
   585→                Добавить ДИ
   586→              </Button>
   587→            </div>
   588→          ) : (
   589→            <div className="overflow-x-auto">
   590→              <Table>
   591→                <TableHeader>
   592→                  <TableRow>
   593→                    <TableHead className="w-[280px]">Название</TableHead>
   594→                    <TableHead>Должность</TableHead>
   595→                    <TableHead>Подразделение</TableHead>
   596→                    <TableHead className="w-[120px]">Дата загрузки</TableHead>
   597→                    <TableHead className="w-[140px] text-right">Действия</TableHead>
   598→                  </TableRow>
   599→                </TableHeader>
   600→                <TableBody>
   601→                  {filteredDIs.map((di) => (
   602→                    <TableRow key={di.id} className="group">
   603→                      <TableCell>
   604→                        <div className="flex items-center gap-2">
   605→                          <FileText className="h-4 w-4 text-amber-600 flex-shrink-0" />
   606→                          <div className="min-w-0">
   607→                            <p className="font-medium truncate">{di.title}</p>
   608→                            {di.fileName && (
   609→                              <p className="text-xs text-muted-foreground truncate">
   610→                                📎 {di.fileName}
   611→                              </p>
   612→                            )}
   613→                          </div>
   614→                        </div>
   615→                      </TableCell>
   616→                      <TableCell>
   617→                        <span className="text-sm">{di.position.title}</span>
   618→                      </TableCell>
   619→                      <TableCell>
   620→                        <Badge variant="outline" className="text-xs">
   621→                          {di.position.department.name}
   622→                        </Badge>
   623→                      </TableCell>
   624→                      <TableCell>
   625→                        <span className="text-sm text-muted-foreground">
   626→                          {formatDate(di.uploadedAt)}
   627→                        </span>
   628→                      </TableCell>
   629→                      <TableCell className="text-right">
   630→                        <div className="flex items-center justify-end gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
