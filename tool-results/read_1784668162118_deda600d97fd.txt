    90→// ========================================
    91→// Department Tree Node
    92→// ========================================
    93→function DepartmentTreeNode({
    94→  dept,
    95→  allDepartments,
    96→  selectedDeptId,
    97→  onSelect,
    98→  expandedIds,
    99→  onToggleExpand,
   100→  onAddChild,
   101→  onEdit,
   102→  onDelete,
   103→  depth = 0,
   104→}: {
   105→  dept: Department
   106→  allDepartments: Department[]
   107→  selectedDeptId: string | null
   108→  onSelect: (dept: Department) => void
   109→  expandedIds: Set<string>
   110→  onToggleExpand: (id: string) => void
   111→  onAddChild: (dept: Department) => void
   112→  onEdit: (dept: Department) => void
   113→  onDelete: (dept: Department) => void
   114→  depth?: number
   115→}) {
   116→  const children = allDepartments.filter((d) => d.parentId === dept.id)
   117→  const hasChildren = children.length > 0
   118→  const isExpanded = expandedIds.has(dept.id)
   119→  const isSelected = selectedDeptId === dept.id
   120→  const posCount = dept._count?.positions ?? 0
   121→
   122→  return (
   123→    <div>
   124→      <div
   125→        className={`flex items-center gap-1 py-1.5 px-2 rounded-md cursor-pointer group transition-colors ${
   126→          isSelected ? 'bg-primary/10 text-primary' : 'hover:bg-muted/60'
   127→        }`}
   128→        style={{ paddingLeft: `${depth * 20 + 8}px` }}
   129→        onClick={() => onSelect(dept)}
   130→      >
   131→        {/* Expand/Collapse Toggle */}
   132→        <button
   133→          className="h-5 w-5 flex items-center justify-center flex-shrink-0"
   134→          onClick={(e) => {
   135→            e.stopPropagation()
   136→            onToggleExpand(dept.id)
   137→          }}
   138→        >
   139→          {hasChildren ? (
   140→            isExpanded ? (
   141→              <ChevronDown className="h-3.5 w-3.5 text-muted-foreground" />
   142→            ) : (
   143→              <ChevronRight className="h-3.5 w-3.5 text-muted-foreground" />
   144→            )
   145→          ) : (
   146→            <span className="h-3.5 w-3.5" />
   147→          )}
   148→        </button>
   149→
   150→        {/* Icon */}
   151→        <Building2 className="h-4 w-4 flex-shrink-0 text-muted-foreground" />
   152→
   153→        {/* Name */}
   154→        <span className="text-sm truncate flex-1">{dept.name}</span>
   155→
   156→        {/* Position count badge */}
   157→        {posCount > 0 && (
   158→          <Badge variant="secondary" className="text-xs h-5 px-1.5 flex-shrink-0">
   159→            {posCount}
   160→          </Badge>
   161→        )}
   162→
   163→        {/* Actions */}
   164→        <div className="hidden group-hover:flex items-center gap-0.5 flex-shrink-0">
   165→          <button
   166→            className="h-6 w-6 flex items-center justify-center rounded hover:bg-muted transition-colors"
   167→            onClick={(e) => {
   168→              e.stopPropagation()
   169→              onAddChild(dept)
   170→            }}
   171→            title="Добавить дочернее"
   172→          >
   173→            <Plus className="h-3 w-3" />
   174→          </button>
   175→          <button
   176→            className="h-6 w-6 flex items-center justify-center rounded hover:bg-muted transition-colors"
   177→            onClick={(e) => {
   178→              e.stopPropagation()
   179→              onEdit(dept)
   180→            }}
   181→            title="Редактировать"
   182→          >
   183→            <Pencil className="h-3 w-3" />
   184→          </button>
   185→          <button
   186→            className="h-6 w-6 flex items-center justify-center rounded hover:bg-destructive/10 hover:text-destructive transition-colors"
   187→            onClick={(e) => {
   188→              e.stopPropagation()
   189→              onDelete(dept)
   190→            }}
   191→            title="Удалить"
   192→          >
   193→            <Trash2 className="h-3 w-3" />
   194→          </button>
   195→        </div>
   196→      </div>
   197→
   198→      {/* Children */}
   199→      {hasChildren && isExpanded && (
   200→        <div>
   201→          {children.map((child) => (
   202→            <DepartmentTreeNode
   203→              key={child.id}
   204→              dept={child}
   205→              allDepartments={allDepartments}
   206→              selectedDeptId={selectedDeptId}
   207→              onSelect={onSelect}
   208→              expandedIds={expandedIds}
   209→              onToggleExpand={onToggleExpand}
   210→              onAddChild={onAddChild}
   211→              onEdit={onEdit}
   212→              onDelete={onDelete}
   213→              depth={depth + 1}
   214→            />
   215→          ))}
   216→        </div>
   217→      )}
   218→    </div>
   219→  )
   220→}

   596→
   597→      for (const line of lines) {
