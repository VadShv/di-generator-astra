     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import { useToast } from '@/hooks/use-toast'
     5→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Textarea } from '@/components/ui/textarea'
     9→import { Label } from '@/components/ui/label'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Switch } from '@/components/ui/switch'
    12→import { Separator } from '@/components/ui/separator'
    13→import {
    14→  AlertDialog,
    15→  AlertDialogAction,
    16→  AlertDialogCancel,
    17→  AlertDialogContent,
    18→  AlertDialogDescription,
    19→  AlertDialogFooter,
    20→  AlertDialogHeader,
    21→  AlertDialogTitle,
    22→} from '@/components/ui/alert-dialog'
    23→import {
    24→  FileText,
    25→  Plus,
    26→  Pencil,
    27→  Trash2,
    28→  Eye,
    29→  ArrowUp,
    30→  ArrowDown,
    31→  ChevronRight,
    32→  GripVertical,
    33→  Copy,
    34→  BookOpen,
    35→  Loader2,
    36→  Save,
    37→  X,
    38→  ListOrdered,
    39→} from 'lucide-react'
    40→
    41→// Types
    42→interface TemplateSection {
    43→  id?: string
    44→  title: string
    45→  order: number
    46→  promptGuidance?: string | null
    47→  isRequired: boolean
    48→  content?: string | null
    49→}
    50→
    51→interface Template {
    52→  id: string
    53→  name: string
    54→  description?: string | null
    55→  isActive: boolean
    56→  sections: TemplateSection[]
    57→  createdAt: string
    58→  updatedAt: string
    59→}
    60→
    61→type ViewMode = 'list' | 'edit' | 'preview'
    62→
    63→const DEFAULT_SECTIONS: Omit<TemplateSection, 'id'>[] = [
    64→  {
    65→    title: 'Общие положения',
    66→    order: 1,
    67→    promptGuidance: 'Опишите общие положения должностной инструкции: назначение документа, область применения, нормативные ссылки, порядок назначения и освобождения от должности, подчинённость и замещение.',
    68→    isRequired: true,
    69→    content: null,
    70→  },
    71→  {
    72→    title: 'Квалификационные требования',
    73→    order: 2,
    74→    promptGuidance: 'Укажите требования к квалификации: образование, опыт работы, профессиональные знания, навыки и компетенции, необходимые для замещения данной должности.',
    75→    isRequired: true,
    76→    content: null,
    77→  },
    78→  {
    79→    title: 'Должностные обязанности',
    80→    order: 3,
    81→    promptGuidance: 'Перечислите должностные обязанности работника: основные задачи, функции, виды работ, которые работник должен выполнять в соответствии с занимаемой должностью.',
    82→    isRequired: true,
    83→    content: null,
    84→  },
    85→  {
    86→    title: 'Права',
    87→    order: 4,
    88→    promptGuidance: 'Опишите права работника: право на информацию, право принятия решений, право подписи документов, право внесения предложений и другие права, необходимые для выполнения должностных обязанностей.',
    89→    isRequired: true,
    90→    content: null,
    91→  },
    92→  {
    93→    title: 'Ответственность',
    94→    order: 5,
    95→    promptGuidance: 'Укажите виды ответственности работника: за невыполнение должностных обязанностей, за правонарушения, за причинение материального ущерба, за нарушение трудовой дисциплины.',
    96→    isRequired: true,
    97→    content: null,
    98→  },
    99→  {
   100→    title: 'Взаимоотношения по должности',
   101→    order: 6,
   102→    promptGuidance: 'Опишите систему взаимодействий: с кем связан по работе, что получает, что передаёт, порядок обмена информацией, сроки и формы представления информации.',
   103→    isRequired: true,
   104→    content: null,
   105→  },
   106→]
   107→
   108→export function TemplatesModule() {
   109→  const { toast } = useToast()
   110→  const [templates, setTemplates] = useState<Template[]>([])
   111→  const [loading, setLoading] = useState(true)
   112→  const [viewMode, setViewMode] = useState<ViewMode>('list')
   113→  const [currentTemplate, setCurrentTemplate] = useState<Partial<Template> | null>(null)
   114→  const [editingSections, setEditingSections] = useState<TemplateSection[]>([])
   115→  const [saving, setSaving] = useState(false)
   116→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   117→  const [templateToDelete, setTemplateToDelete] = useState<string | null>(null)
   118→  const [previewTemplate, setPreviewTemplate] = useState<Template | null>(null)
   119→
   120→  // Fetch templates
   121→  const fetchTemplates = useCallback(async () => {
   122→    try {
   123→      const res = await fetch('/api/templates')
   124→      if (res.ok) {
   125→        const data = await res.json()
   126→        setTemplates(data)
   127→      }
   128→    } catch (e) {
   129→      console.error(e)
   130→      toast({ title: 'Ошибка', description: 'Не удалось загрузить шаблоны', variant: 'destructive' })
   131→    } finally {
   132→      setLoading(false)
   133→    }
   134→  }, [toast])
   135→
   136→  useEffect(() => {
   137→    fetchTemplates()
   138→  }, [fetchTemplates])
   139→
   140→  // Create new template
   141→  const handleCreateNew = () => {
   142→    setCurrentTemplate({
   143→      name: '',
   144→      description: '',
   145→      isActive: true,
   146→    })
   147→    setEditingSections([])
   148→    setViewMode('edit')
   149→  }
   150→
   151→  // Create from default Russian template
   152→  const handleCreateDefault = () => {
   153→    setCurrentTemplate({
   154→      name: 'Стандартный шаблон ДИ',
   155→      description: 'Стандартный шаблон должностной инструкции с типовыми разделами',
   156→      isActive: true,
   157→    })
   158→    setEditingSections(
   159→      DEFAULT_SECTIONS.map((s) => ({
   160→        ...s,
   161→        id: `new-${Date.now()}-${s.order}`,
   162→      }))
   163→    )
   164→    setViewMode('edit')
   165→  }
   166→
   167→  // Edit existing template
   168→  const handleEdit = (template: Template) => {
   169→    setCurrentTemplate({
   170→      id: template.id,
   171→      name: template.name,
   172→      description: template.description,
   173→      isActive: template.isActive,
   174→    })
   175→    setEditingSections([...template.sections])
   176→    setViewMode('edit')
   177→  }
   178→
   179→  // Preview template
   180→  const handlePreview = (template: Template) => {
   181→    setPreviewTemplate(template)
   182→    setViewMode('preview')
   183→  }
   184→
   185→  // Duplicate template
   186→  const handleDuplicate = async (template: Template) => {
   187→    try {
   188→      setSaving(true)
   189→      const res = await fetch('/api/templates', {
   190→        method: 'POST',
   191→        headers: { 'Content-Type': 'application/json' },
   192→        body: JSON.stringify({
   193→          name: `${template.name} (копия)`,
   194→          description: template.description,
   195→          sections: template.sections.map((s) => ({
   196→            title: s.title,
   197→            order: s.order,
   198→            promptGuidance: s.promptGuidance,
   199→            isRequired: s.isRequired,
   200→            content: s.content,
   201→          })),
   202→        }),
   203→      })
   204→      if (res.ok) {
   205→        toast({ title: 'Шаблон скопирован', description: 'Создана копия шаблона' })
   206→        fetchTemplates()
   207→      } else {
   208→        const data = await res.json()
   209→        toast({ title: 'Ошибка', description: data.error || 'Не удалось скопировать', variant: 'destructive' })
   210→      }
   211→    } catch {
   212→      toast({ title: 'Ошибка', description: 'Не удалось скопировать шаблон', variant: 'destructive' })
   213→    } finally {
   214→      setSaving(false)
   215→    }
   216→  }
   217→
   218→  // Delete template
   219→  const handleDeleteConfirm = async () => {
   220→    if (!templateToDelete) return
   221→    try {
   222→      const res = await fetch('/api/templates', {
   223→        method: 'DELETE',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({ id: templateToDelete }),
   226→      })
   227→      if (res.ok) {
   228→        toast({ title: 'Удалено', description: 'Шаблон успешно удалён' })
   229→        fetchTemplates()
   230→      } else {
   231→        const data = await res.json()
   232→        toast({ title: 'Ошибка', description: data.error || 'Не удалось удалить', variant: 'destructive' })
   233→      }
   234→    } catch {
   235→      toast({ title: 'Ошибка', description: 'Не удалось удалить шаблон', variant: 'destructive' })
   236→    } finally {
   237→      setDeleteDialogOpen(false)
   238→      setTemplateToDelete(null)
   239→    }
   240→  }
   241→
   242→  // Save template (create or update)
   243→  const handleSave = async () => {
   244→    if (!currentTemplate?.name?.trim()) {
   245→      toast({ title: 'Ошибка', description: 'Укажите название шаблона', variant: 'destructive' })
   246→      return
   247→    }
   248→    if (editingSections.length === 0) {
   249→      toast({ title: 'Ошибка', description: 'Добавьте хотя бы одну секцию', variant: 'destructive' })
   250→      return
   251→    }
   252→    // Validate section titles
   253→    for (const s of editingSections) {
   254→      if (!s.title.trim()) {
   255→        toast({ title: 'Ошибка', description: 'Все секции должны иметь название', variant: 'destructive' })
   256→        return
   257→      }
   258→    }
   259→
   260→    setSaving(true)
   261→    try {
   262→      const payload = {
   263→        ...(currentTemplate.id ? { id: currentTemplate.id } : {}),
   264→        name: currentTemplate.name,
   265→        description: currentTemplate.description,
   266→        isActive: currentTemplate.isActive,
   267→        sections: editingSections.map((s, idx) => ({
   268→          title: s.title,
   269→          order: idx,
   270→          promptGuidance: s.promptGuidance,
   271→          isRequired: s.isRequired,
   272→          content: s.content,
   273→        })),
   274→      }
   275→
   276→      const method = currentTemplate.id ? 'PUT' : 'POST'
   277→      const res = await fetch('/api/templates', {
   278→        method,
   279→        headers: { 'Content-Type': 'application/json' },
   280→        body: JSON.stringify(payload),
   281→      })
   282→
   283→      if (res.ok) {
   284→        toast({
   285→          title: currentTemplate.id ? 'Шаблон обновлён' : 'Шаблон создан',
   286→          description: currentTemplate.id ? 'Изменения сохранены' : 'Новый шаблон добавлен',
   287→        })
   288→        setViewMode('list')
   289→        fetchTemplates()
   290→      } else {
   291→        const data = await res.json()
   292→        toast({ title: 'Ошибка', description: data.error || 'Не удалось сохранить', variant: 'destructive' })
   293→      }
   294→    } catch {
   295→      toast({ title: 'Ошибка', description: 'Не удалось сохранить шаблон', variant: 'destructive' })
   296→    } finally {
   297→      setSaving(false)
   298→    }
   299→  }
   300→
   301→  // Section management helpers
   302→  const addSection = () => {
   303→    const newSection: TemplateSection = {
   304→      id: `new-${Date.now()}`,
   305→      title: '',
   306→      order: editingSections.length,
   307→      promptGuidance: null,
   308→      isRequired: false,
   309→      content: null,
   310→    }
   311→    setEditingSections([...editingSections, newSection])
   312→  }
   313→
   314→  const removeSection = (index: number) => {
   315→    setEditingSections(editingSections.filter((_, i) => i !== index))
   316→  }
   317→
   318→  const moveSectionUp = (index: number) => {
   319→    if (index === 0) return
   320→    const newSections = [...editingSections]
   321→    const temp = newSections[index]
   322→    newSections[index] = newSections[index - 1]
   323→    newSections[index - 1] = temp
   324→    setEditingSections(newSections)
   325→  }
   326→
   327→  const moveSectionDown = (index: number) => {
   328→    if (index === editingSections.length - 1) return
   329→    const newSections = [...editingSections]
   330→    const temp = newSections[index]
   331→    newSections[index] = newSections[index + 1]
   332→    newSections[index + 1] = temp
   333→    setEditingSections(newSections)
   334→  }
   335→
   336→  const updateSection = (index: number, field: keyof TemplateSection, value: string | boolean | null) => {
   337→    const newSections = [...editingSections]
   338→    newSections[index] = { ...newSections[index], [field]: value }
   339→    setEditingSections(newSections)
   340→  }
   341→
   342→  // Cancel editing
   343→  const handleCancel = () => {
   344→    setCurrentTemplate(null)
   345→    setEditingSections([])
   346→    setViewMode('list')
   347→  }
   348→
   349→  // ==================== RENDER: LIST VIEW ====================
   350→  if (viewMode === 'list') {
   351→    return (
   352→      <div className="space-y-6">
   353→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   354→          <div>
   355→            <h1 className="text-3xl font-bold tracking-tight">Шаблоны ДИ</h1>
   356→            <p className="text-muted-foreground mt-1">
   357→              Управление шаблонами должностных инструкций
   358→            </p>
   359→          </div>
   360→          <div className="flex gap-2">
   361→            <Button variant="outline" onClick={handleCreateDefault} className="gap-2">
   362→              <BookOpen className="h-4 w-4" />
   363→              Стандартный шаблон
   364→            </Button>
   365→            <Button onClick={handleCreateNew} className="gap-2">
   366→              <Plus className="h-4 w-4" />
   367→              Новый шаблон
   368→            </Button>
   369→          </div>
   370→        </div>
   371→
   372→        {loading ? (
   373→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   374→            {Array.from({ length: 3 }).map((_, i) => (
   375→              <Card key={i} className="animate-pulse">
   376→                <CardHeader className="p-6">
   377→                  <div className="h-5 bg-muted rounded w-3/4" />
   378→                  <div className="h-4 bg-muted rounded w-1/2 mt-2" />
   379→                </CardHeader>
   380→                <CardContent className="px-6 pb-6">
   381→                  <div className="h-4 bg-muted rounded w-1/3" />
   382→                </CardContent>
   383→              </Card>
   384→            ))}
   385→          </div>
   386→        ) : templates.length === 0 ? (
   387→          <Card className="border-dashed">
   388→            <CardContent className="flex flex-col items-center justify-center py-16">
   389→              <FileText className="h-12 w-12 text-muted-foreground/50 mb-4" />
   390→              <h3 className="text-lg font-semibold mb-1">Нет шаблонов</h3>
   391→              <p className="text-muted-foreground text-sm mb-6 text-center max-w-md">
   392→                Создайте первый шаблон должностной инструкции или используйте стандартный шаблон с типовыми разделами
   393→              </p>
   394→              <div className="flex gap-2">
   395→                <Button variant="outline" onClick={handleCreateDefault} className="gap-2">
   396→                  <BookOpen className="h-4 w-4" />
   397→                  Стандартный шаблон
   398→                </Button>
   399→                <Button onClick={handleCreateNew} className="gap-2">
   400→                  <Plus className="h-4 w-4" />
   401→                  Новый шаблон
   402→                </Button>
   403→              </div>
   404→            </CardContent>
   405→          </Card>
   406→        ) : (
   407→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   408→            {templates.map((template) => (
   409→              <Card
   410→                key={template.id}
   411→                className="hover:shadow-md transition-shadow group"
   412→              >
   413→                <CardHeader className="p-6 pb-3">
   414→                  <div className="flex items-start justify-between gap-2">
   415→                    <CardTitle className="text-base leading-tight">{template.name}</CardTitle>
   416→                    <Badge variant={template.isActive ? 'default' : 'secondary'} className="shrink-0 text-xs">
   417→                      {template.isActive ? 'Активен' : 'Неактивен'}
   418→                    </Badge>
   419→                  </div>
   420→                  {template.description && (
   421→                    <CardDescription className="mt-1 line-clamp-2">
   422→                      {template.description}
   423→                    </CardDescription>
   424→                  )}
   425→                </CardHeader>
   426→                <CardContent className="px-6 pb-4 pt-0">
   427→                  <div className="flex items-center gap-1.5 text-sm text-muted-foreground mb-4">
   428→                    <ListOrdered className="h-3.5 w-3.5" />
   429→                    <span>{template.sections.length} {getSectionsWord(template.sections.length)}</span>
   430→                  </div>
   431→                  <Separator className="mb-3" />
   432→                  <div className="flex items-center gap-1">
   433→                    <Button
   434→                      variant="ghost"
   435→                      size="sm"
   436→                      className="h-8 gap-1.5 text-xs flex-1"
   437→                      onClick={() => handlePreview(template)}
   438→                    >
   439→                      <Eye className="h-3.5 w-3.5" />
   440→                      Просмотр
   441→                    </Button>
   442→                    <Button
   443→                      variant="ghost"
   444→                      size="sm"
   445→                      className="h-8 gap-1.5 text-xs flex-1"
   446→                      onClick={() => handleEdit(template)}
   447→                    >
   448→                      <Pencil className="h-3.5 w-3.5" />
   449→                      Изменить
   450→                    </Button>
   451→                    <Button
   452→                      variant="ghost"
   453→                      size="sm"
   454→                      className="h-8 gap-1.5 text-xs"
   455→                      onClick={() => handleDuplicate(template)}
   456→                      title="Копировать"
   457→                    >
   458→                      <Copy className="h-3.5 w-3.5" />
   459→                    </Button>
   460→                    <Button
   461→                      variant="ghost"
   462→                      size="sm"
   463→                      className="h-8 gap-1.5 text-xs text-destructive hover:text-destructive"
   464→                      onClick={() => {
   465→                        setTemplateToDelete(template.id)
   466→                        setDeleteDialogOpen(true)
   467→                      }}
   468→                      title="Удалить"
   469→                    >
   470→                      <Trash2 className="h-3.5 w-3.5" />
   471→                    </Button>
   472→                  </div>
   473→                </CardContent>
   474→              </Card>
   475→            ))}
   476→          </div>
   477→        )}
   478→
   479→        {/* Delete confirmation dialog */}
   480→        <AlertDialog open={deleteDialogOpen} onOpenChange={setDeleteDialogOpen}>
   481→          <AlertDialogContent>
   482→            <AlertDialogHeader>
   483→              <AlertDialogTitle>Удалить шаблон?</AlertDialogTitle>
   484→              <AlertDialogDescription>
   485→                Это действие нельзя отменить. Шаблон и все его секции будут удалены.
   486→              </AlertDialogDescription>
   487→            </AlertDialogHeader>
   488→            <AlertDialogFooter>
   489→              <AlertDialogCancel>Отмена</AlertDialogCancel>
   490→              <AlertDialogAction onClick={handleDeleteConfirm} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">
   491→                Удалить
   492→              </AlertDialogAction>
   493→            </AlertDialogFooter>
   494→          </AlertDialogContent>
   495→        </AlertDialog>
   496→      </div>
   497→    )
   498→  }
   499→
   500→  // ==================== RENDER: EDIT VIEW ====================
   501→  if (viewMode === 'edit') {
   502→    return (
   503→      <div className="space-y-6">
   504→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   505→          <div>
   506→            <h1 className="text-3xl font-bold tracking-tight">
   507→              {currentTemplate?.id ? 'Редактирование шаблона' : 'Новый шаблон'}
   508→            </h1>
   509→            <p className="text-muted-foreground mt-1">
   510→              {currentTemplate?.id ? 'Измените параметры шаблона и секции' : 'Заполните данные и добавьте секции шаблона'}
   511→            </p>
   512→          </div>
   513→          <div className="flex gap-2">
   514→            <Button variant="outline" onClick={handleCancel} className="gap-2">
   515→              <X className="h-4 w-4" />
   516→              Отмена
   517→            </Button>
   518→            <Button onClick={handleSave} disabled={saving} className="gap-2">
   519→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}
   520→              Сохранить
   521→            </Button>
   522→          </div>
   523→        </div>
   524→
   525→        {/* Basic Info Card */}
   526→        <Card>
   527→          <CardHeader className="p-6 pb-4">
   528→            <CardTitle className="text-base">Основная информация</CardTitle>
   529→          </CardHeader>
   530→          <CardContent className="px-6 pb-6 space-y-4">
   531→            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   532→              <div className="space-y-2">
   533→                <Label htmlFor="template-name">Название шаблона *</Label>
   534→                <Input
   535→                  id="template-name"
   536→                  placeholder="Например: Стандартный шаблон ДИ"
   537→                  value={currentTemplate?.name || ''}
   538→                  onChange={(e) =>
   539→                    setCurrentTemplate({ ...currentTemplate!, name: e.target.value })
   540→                  }
   541→                />
   542→              </div>
   543→              <div className="space-y-2">
   544→                <Label htmlFor="template-desc">Описание</Label>
   545→                <Input
   546→                  id="template-desc"
   547→                  placeholder="Краткое описание шаблона"
   548→                  value={currentTemplate?.description || ''}
   549→                  onChange={(e) =>
   550→                    setCurrentTemplate({ ...currentTemplate!, description: e.target.value })
   551→                  }
   552→                />
   553→              </div>
   554→            </div>
   555→            <div className="flex items-center gap-3">
   556→              <Switch
   557→                id="template-active"
   558→                checked={currentTemplate?.isActive ?? true}
   559→                onCheckedChange={(checked) =>
   560→                  setCurrentTemplate({ ...currentTemplate!, isActive: checked })
   561→                }
   562→              />
   563→              <Label htmlFor="template-active">Шаблон активен</Label>
   564→            </div>
   565→          </CardContent>
   566→        </Card>
   567→
   568→        {/* Sections Card */}
   569→        <Card>
   570→          <CardHeader className="p-6 pb-4">
   571→            <div className="flex items-center justify-between">
   572→              <div>
   573→                <CardTitle className="text-base">Секции шаблона</CardTitle>
   574→                <CardDescription className="mt-1">
   575→                  Определите структуру должностной инструкции. Порядок секций определяет их расположение в итоговом документе.
   576→                </CardDescription>
   577→              </div>
   578→              <Button variant="outline" size="sm" onClick={addSection} className="gap-2 shrink-0">
   579→                <Plus className="h-4 w-4" />
   580→                Добавить секцию
   581→              </Button>
   582→            </div>
   583→          </CardHeader>
   584→          <CardContent className="px-6 pb-6">
   585→            {editingSections.length === 0 ? (
   586→              <div className="flex flex-col items-center py-8 text-center">
   587→                <ListOrdered className="h-8 w-8 text-muted-foreground/50 mb-3" />
   588→                <p className="text-sm text-muted-foreground mb-3">

   592→                  <Plus className="h-4 w-4" />
   593→                  Добавить секцию
   594→                </Button>
   595→              </div>

   598→                {editingSections.map((section, idx) => (
   599→                  <SectionEditor
   600→                    key={section.id || `section-${idx}`}
   601→                    section={section}
   602→                    index={idx}
   603→                    total={editingSections.length}
   604→                    onUpdate={(field, value) => updateSection(idx, field, value)}
   605→                    onRemove={() => removeSection(idx)}
   606→                    onMoveUp={() => moveSectionUp(idx)}
   607→                    onMoveDown={() => moveSectionDown(idx)}
   608→                  />
   609→                ))}
   610→              </div>
   611→            )}
   612→          </CardContent>
   613→        </Card>
   614→
   615→        {/* Quick add default sections */}
   616→        {editingSections.length === 0 && (
   617→          <Card className="border-dashed">
   618→            <CardContent className="flex flex-col items-center py-8">
   619→              <BookOpen className="h-8 w-8 text-muted-foreground/50 mb-3" />

   621→                Или добавьте стандартные секции ДИ одним нажатием

   624→                variant="outline"
   625→                size="sm"
   626→                onClick={() => {
   627→                  setEditingSections(
   628→                    DEFAULT_SECTIONS.map((s) => ({
   629→                      ...s,
   630→                      id: `new-${Date.now()}-${s.order}`,

   633→                }}
   634→                className="gap-2"

   636→                <BookOpen className="h-4 w-4" />

   638→              </Button>
   639→            </CardContent>

   641→        )}
   642→      </div>
   643→    )
   644→  }
   645→
   646→  // ==================== RENDER: PREVIEW VIEW ====================
   647→  if (viewMode === 'preview' && previewTemplate) {
   648→    return (
   649→      <div className="space-y-6">
   650→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

   653→            <p className="text-muted-foreground mt-1">
   654→              Как будет выглядеть документ на основе этого шаблона
   655→            </p>
   656→          </div>

   660→              Закрыть
   661→            </Button>
   662→            <Button
   663→              variant="outline"
   664→              onClick={() => {
   665→                handleEdit(previewTemplate)
   666→              }}
   667→              className="gap-2"
   668→            >
   669→              <Pencil className="h-4 w-4" />
   670→              Редактировать
   671→            </Button>

   678→              <div>

   680→                {previewTemplate.description && (
   681→                  <CardDescription className="mt-1">{previewTemplate.description}</CardDescription>
   682→                )}
   683→              </div>
   684→              <Badge variant={previewTemplate.isActive ? 'default' : 'secondary'}>
   685→                {previewTemplate.isActive ? 'Активен' : 'Неактивен'}
   686→              </Badge>
   687→            </div>
   688→          </CardHeader>
   689→          <CardContent className="px-6 pb-6">
   690→            <Separator className="mb-6" />
   691→            <div className="space-y-6">
   692→              {previewTemplate.sections
   693→                .sort((a, b) => a.order - b.order)
   694→                .map((section, idx) => (
   695→                  <div key={section.id || idx} className="space-y-2">
   696→                    <h3 className="text-lg font-semibold flex items-center gap-2">
   697→                      <span className="text-primary font-bold">{idx + 1}.</span>
   698→                      {section.title}
   699→                      {section.isRequired && (
   700→                        <Badge variant="outline" className="text-xs font-normal">
   701→                          Обязательно
   702→                        </Badge>
   703→                      )}
   704→                    </h3>
   705→                    {section.promptGuidance && (
   706→                      <div className="bg-muted/50 rounded-lg p-4 text-sm text-muted-foreground border-l-4 border-primary/30">
   707→                        <p className="font-medium text-foreground/80 mb-1 flex items-center gap-1.5">
   708→                          <BookOpen className="h-3.5 w-3.5" />
   709→                          Руководство для ИИ:
   710→                        </p>
   711→                        {section.promptGuidance}
   712→                      </div>
   713→                    )}
   714→                    {section.content && (
   715→                      <div className="text-sm bg-muted/30 rounded-lg p-4 whitespace-pre-wrap">
   716→                        {section.content}
   717→                      </div>
   718→                    )}
   719→                    {!section.content && !section.promptGuidance && (
   720→                      <p className="text-sm text-muted-foreground italic">
   721→                        Содержимое будет сгенерировано ИИ на основе данных должности
   722→                      </p>
   723→                    )}
   724→                    {idx < previewTemplate.sections.length - 1 && <Separator className="mt-4" />}
   725→                  </div>
   726→                ))}
   727→            </div>
   728→          </CardContent>
   729→        </Card>
   730→      </div>
   731→    )
   732→  }
   733→
   734→  return null
   735→}

   738→interface SectionEditorProps {

   748→function SectionEditor({ section, index, total, onUpdate, onRemove, onMoveUp, onMoveDown }: SectionEditorProps) {

   859→// Helper for Russian word forms
   860→function getSectionsWord(count: number): string {
   861→  const lastTwo = count % 100
   862→  const lastOne = count % 10
   863→  if (lastTwo >= 11 && lastTwo <= 19) return 'секций'
   864→  if (lastOne === 1) return 'секция'
   865→  if (lastOne >= 2 && lastOne <= 4) return 'секции'
   866→  return 'секций'
   867→}
