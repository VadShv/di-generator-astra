   488→  const computeDiff = (left: string, right: string) => {
   489→    const leftLines = left.split('\n')
   490→    const rightLines = right.split('\n')
   491→    const maxLen = Math.max(leftLines.length, rightLines.length)
   492→    const result: { left: string; right: string; type: 'same' | 'added' | 'removed' | 'changed' }[] = []
   493→
   494→    for (let i = 0; i < maxLen; i++) {
   495→      const lLine = i < leftLines.length ? leftLines[i] : undefined
   496→      const rLine = i < rightLines.length ? rightLines[i] : undefined
   497→
   498→      if (lLine === rLine) {
   499→        result.push({ left: lLine || '', right: rLine || '', type: 'same' })
   500→      } else if (lLine === undefined) {
   501→        result.push({ left: '', right: rLine || '', type: 'added' })
   502→      } else if (rLine === undefined) {
   503→        result.push({ left: lLine, right: '', type: 'removed' })
   504→      } else {
   505→        result.push({ left: lLine, right: rLine, type: 'changed' })
   506→      }
   507→    }
   508→    return result
   509→  }
