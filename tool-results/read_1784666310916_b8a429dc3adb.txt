   130→function getStatusConfig(status: string) {

   134→function formatDate(dateStr: string) {
   135→  try {
   136→    return new Date(dateStr).toLocaleDateString('ru-RU', {
   137→      day: '2-digit',
   138→      month: '2-digit',
   139→      year: 'numeric',
   140→      hour: '2-digit',
   141→      minute: '2-digit',
   142→    })
   143→  } catch {
   144→    return dateStr
   145→  }
   146→}
   147→
   148→function formatDateShort(dateStr: string) {
   149→  try {
   150→    return new Date(dateStr).toLocaleDateString('ru-RU', {
   151→      day: '2-digit',
   152→      month: '2-digit',
   153→    })
   154→  } catch {
   155→    return dateStr
   156→  }
   157→}
   158→
   159→function daysBetween(dateA: string, dateB: string) {
   160→  const a = new Date(dateA).getTime()
   161→  const b = new Date(dateB).getTime()
   162→  return Math.round(Math.abs(b - a) / (1000 * 60 * 60 * 24))
   163→}
   164→
   165→function daysSince(dateStr: string) {
   166→  return Math.round((Date.now() - new Date(dateStr).getTime()) / (1000 * 60 * 60 * 24))
   167→}
