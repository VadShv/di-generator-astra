     5→import { Button } from '@/components/ui/button'
     6→import { Input } from '@/components/ui/input'
     7→import { Label } from '@/components/ui/label'

    13→} from '@/components/ui/table'

    26→
    27→interface Department { id: string; name: string; code: string; parentId: string | null; parent?: Department | null; children?: Department[]; _count?: { positions: number }; createdAt: string; updatedAt: string }
    28→interface Position { id: string; title: string; code: string; departmentId: string; department: Department; grade: string | null; domain: string | null; headcount: number; functions: string | null; createdAt: string; updatedAt: string }
    29→
    30→export function StaffScheduleModule() {
    31→  const { toast } = useToast()
    32→  const [departments, setDepartments] = useState<Department[]>([])
    33→  const [positions, setPositions] = useState<Position[]>([])
    34→  const [loading, setLoading] = useState(true)
    35→  const [expandedIds, setExpandedIds] = useState<Set<string>>(new Set())
    36→  const [selectedDeptId, setSelectedDeptId] = useState<string | null>(null)
    37→
    38→  // Dept dialog
    39→  const [deptDialogOpen, setDeptDialogOpen] = useState(false)
    40→  const [deptDialogMode, setDeptDialogMode] = useState<'create' | 'edit'>('create')
    41→  const [deptForm, setDeptForm] = useState({ id: '', name: '', code: '', parentId: '' })
    42→  const [deptSubmitting, setDeptSubmitting] = useState(false)
    43→  const [deptDeleteOpen, setDeptDeleteOpen] = useState(false)
    44→  const [deptToDelete, setDeptToDelete] = useState<Department | null>(null)
    45→
    46→  // Pos dialog
    47→  const [posDialogOpen, setPosDialogOpen] = useState(false)
    48→  const [posDialogMode, setPosDialogMode] = useState<'create' | 'edit'>('create')
    49→  const [posForm, setPosForm] = useState({ id: '', title: '', code: '', departmentId: '', grade: '', domain: '', headcount: 1, functions: '' })
    50→  const [posSubmitting, setPosSubmitting] = useState(false)
    51→  const [posDeleteOpen, setPosDeleteOpen] = useState(false)
    52→  const [posToDelete, setPosToDelete] = useState<Position | null>(null)
    53→
    54→  // Bulk text
    55→  const [bulkDialogOpen, setBulkDialogOpen] = useState(false)
    56→  const [bulkText, setBulkText] = useState('')
    57→  const [bulkSubmitting, setBulkSubmitting] = useState(false)
    58→
    59→  // File upload
    60→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false)
    61→  const [uploading, setUploading] = useState(false)
    62→  const [uploadProgress, setUploadProgress] = useState(0)
    63→  const [uploadResult, setUploadResult] = useState<{
    64→    success: boolean
    65→    summary?: { departmentsFound: number; departmentsCreated: number; departmentsExisting: number; positionsFound: number; positionsCreated: number; positionsSkipped: number }
    66→    errors?: string[]
    67→  } | null>(null)
    68→  const [dragActive, setDragActive] = useState(false)
    69→  const [selectedFile, setSelectedFile] = useState<File | null>(null)
    70→  const fileInputRef = useRef<HTMLInputElement>(null)
    71→
    72→  // Filters
    73→  const [searchQuery, setSearchQuery] = useState('')
    74→
    75→  const fetchDepartments = useCallback(async () => {
    76→    try { const res = await fetch('/api/departments'); if (res.ok) setDepartments(await res.json()) } catch { /* silent */ }
    77→  }, [])
    78→  const fetchPositions = useCallback(async () => {
    79→    try { const res = await fetch('/api/positions'); if (res.ok) setPositions(await res.json()) } catch { /* silent */ }
    80→  }, [])
    81→
    82→  useEffect(() => { (async () => { setLoading(true); await Promise.all([fetchDepartments(), fetchPositions()]); setLoading(false) })() }, [fetchDepartments, fetchPositions])
    83→
    84→  const rootDepts = departments.filter(d => !d.parentId)
    85→  const toggleExpand = (id: string) => setExpandedIds(prev => { const n = new Set(prev); if (n.has(id)) { n.delete(id) } else { n.add(id) }; return n })
    86→  const getChildren = (id: string) => departments.filter(d => d.parentId === id)
    87→  const getPosCount = (id: string) => departments.find(d => d.id === id)?._count?.positions ?? 0
    88→
    89→  const filteredPositions = positions.filter(p => {
    90→    if (selectedDeptId && p.departmentId !== selectedDeptId) return false
    91→    if (searchQuery) { const q = searchQuery.toLowerCase(); return p.title.toLowerCase().includes(q) || p.code.toLowerCase().includes(q) || p.department?.name?.toLowerCase().includes(q) || (p.grade || '').toLowerCase().includes(q) || (p.domain || '').toLowerCase().includes(q) }
    92→    return true
    93→  })
    94→
    95→  // Dept handlers
    96→  const openCreateDept = (parentId?: string) => { setDeptDialogMode('create'); setDeptForm({ id: '', name: '', code: '', parentId: parentId || '' }); setDeptDialogOpen(true) }
    97→  const openEditDept = (d: Department) => { setDeptDialogMode('edit'); setDeptForm({ id: d.id, name: d.name, code: d.code, parentId: d.parentId || '' }); setDeptDialogOpen(true) }
    98→
    99→  const handleDeptSubmit = async () => {
   100→    if (!deptForm.name.trim() || !deptForm.code.trim()) { toast({ title: 'Ошибка', description: 'Название и код обязательны', variant: 'destructive' }); return }
   101→    setDeptSubmitting(true)
   102→    try {
   103→      const body: Record<string, string> = { name: deptForm.name.trim(), code: deptForm.code.trim() }
   104→      if (deptForm.parentId) body.parentId = deptForm.parentId
   105→      if (deptDialogMode === 'edit') body.id = deptForm.id
   106→      const res = await fetch('/api/departments', { method: deptDialogMode === 'create' ? 'POST' : 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   107→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Ошибка') }
   108→      toast({ title: deptDialogMode === 'create' ? 'Создано' : 'Обновлено', description: deptForm.name })
   109→      setDeptDialogOpen(false); await fetchDepartments()
   110→    } catch (e) { toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Ошибка', variant: 'destructive' }) }
   111→    finally { setDeptSubmitting(false) }
   112→  }
   113→
   114→  const handleDeptDelete = async () => {
   115→    if (!deptToDelete) return
   116→    try {
   117→      const res = await fetch('/api/departments', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: deptToDelete.id }) })
   118→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Ошибка') }
   119→      toast({ title: 'Удалено', description: deptToDelete.name })
   120→      if (selectedDeptId === deptToDelete.id) setSelectedDeptId(null)
   121→      setDeptDeleteOpen(false); setDeptToDelete(null); await fetchDepartments()
   122→    } catch (e) { toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Ошибка', variant: 'destructive' }) }
   123→  }
   124→
   125→  // Position handlers
   126→  const openCreatePos = (departmentId?: string) => { setPosDialogMode('create'); setPosForm({ id: '', title: '', code: '', departmentId: departmentId || selectedDeptId || '', grade: '', domain: '', headcount: 1, functions: '' }); setPosDialogOpen(true) }
   127→  const openEditPos = (p: Position) => { setPosDialogMode('edit'); setPosForm({ id: p.id, title: p.title, code: p.code, departmentId: p.departmentId, grade: p.grade || '', domain: p.domain || '', headcount: p.headcount, functions: p.functions || '' }); setPosDialogOpen(true) }
   128→
   129→  const handlePosSubmit = async () => {
   130→    if (!posForm.title.trim() || !posForm.code.trim() || !posForm.departmentId) { toast({ title: 'Ошибка', description: 'Название, код и подразделение обязательны', variant: 'destructive' }); return }
   131→    setPosSubmitting(true)
   132→    try {
   133→      const body: Record<string, unknown> = { title: posForm.title.trim(), code: posForm.code.trim(), departmentId: posForm.departmentId, grade: posForm.grade || null, domain: posForm.domain || null, headcount: posForm.headcount || 1, functions: posForm.functions || null }
   134→      if (posDialogMode === 'edit') body.id = posForm.id
   135→      const res = await fetch('/api/positions', { method: posDialogMode === 'create' ? 'POST' : 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   136→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Ошибка') }
   137→      toast({ title: posDialogMode === 'create' ? 'Создано' : 'Обновлено', description: posForm.title })
   138→      setPosDialogOpen(false); await fetchPositions(); await fetchDepartments()
   139→    } catch (e) { toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Ошибка', variant: 'destructive' }) }
   140→    finally { setPosSubmitting(false) }
   141→  }
   142→
   143→  const handlePosDelete = async () => {
   144→    if (!posToDelete) return
   145→    try {
   146→      const res = await fetch('/api/positions', { method: 'DELETE', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: posToDelete.id }) })
   147→      if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Ошибка') }
   148→      toast({ title: 'Удалено', description: posToDelete.title })
   149→      setPosDeleteOpen(false); setPosToDelete(null); await fetchPositions(); await fetchDepartments()
   150→    } catch (e) { toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Ошибка', variant: 'destructive' }) }
   151→  }
   152→
   153→  const handleBulkUpload = async () => {
   154→    if (!bulkText.trim()) { toast({ title: 'Ошибка', description: 'Введите данные', variant: 'destructive' }); return }
   155→    setBulkSubmitting(true)
   156→    try {
   157→      const lines = bulkText.trim().split('\n').filter(l => l.trim()); let created = 0; let errors = 0
   158→      for (const line of lines) {
   159→        const [code, title, deptCode, grade, domain, headcountStr, functions] = line.split(';').map(s => s.trim())
   160→        if (!code || !title || !deptCode) { errors++; continue }
   161→        const dept = departments.find(d => d.code === deptCode)
   162→        if (!dept || positions.find(p => p.code === code)) { errors++; continue }
   163→        try {
   164→          const res = await fetch('/api/positions', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code, title, departmentId: dept.id, grade: grade || null, domain: domain || null, headcount: headcountStr ? parseInt(headcountStr, 10) || 1 : 1, functions: functions || null }) })
   165→          if (res.ok) { created++ } else { errors++ }
   166→        } catch { errors++ }
   167→      }
   168→      toast({ title: 'Загрузка завершена', description: `Создано: ${created}, Ошибок: ${errors}` })
   169→      setBulkDialogOpen(false); setBulkText(''); await fetchPositions(); await fetchDepartments()
   170→    } catch { toast({ title: 'Ошибка', description: 'Ошибка загрузки', variant: 'destructive' }) }
   171→    finally { setBulkSubmitting(false) }
   172→  }
   173→
   174→  // File upload handler
   175→  const handleFileUpload = async () => {
   176→    if (!selectedFile) { toast({ title: 'Ошибка', description: 'Выберите файл', variant: 'destructive' }); return }
   177→    setUploading(true); setUploadProgress(10); setUploadResult(null)
   178→    try {
   179→      const formData = new FormData()
   180→      formData.append('file', selectedFile)
   181→      setUploadProgress(30)
   182→      
   183→      const res = await fetch('/api/upload/staff-schedule', { method: 'POST', body: formData })
   184→      setUploadProgress(80)
   185→      
   186→      if (!res.ok) {
   187→        const err = await res.json()
   188→        throw new Error(err.error || 'Ошибка загрузки')
   189→      }
   190→      
   191→      const data = await res.json()
   192→      setUploadProgress(100)
   193→      setUploadResult(data)
   194→      
   195→      if (data.success) {
   196→        toast({ 
   197→          title: 'Файл обработан', 
   198→          description: `Подразделений: ${data.summary.departmentsCreated} создано, Должностей: ${data.summary.positionsCreated} создано` 
   199→        })
   200→        await fetchPositions(); await fetchDepartments()
   201→      }
   202→    } catch (e) {
   203→      toast({ title: 'Ошибка загрузки файла', description: e instanceof Error ? e.message : 'Ошибка', variant: 'destructive' })
   204→    } finally {
   205→      setUploading(false)
   206→    }
   207→  }
   208→
   209→  const handleDrag = useCallback((e: React.DragEvent) => {
   210→    e.preventDefault(); e.stopPropagation()
   211→    if (e.type === 'dragenter' || e.type === 'dragover') setDragActive(true)
   212→    else if (e.type === 'dragleave') setDragActive(false)
   213→  }, [])
   214→
   215→  const handleDrop = useCallback((e: React.DragEvent) => {
   216→    e.preventDefault(); e.stopPropagation(); setDragActive(false)
   217→    if (e.dataTransfer.files && e.dataTransfer.files[0]) {
   218→      setSelectedFile(e.dataTransfer.files[0])
   219→    }
   220→  }, [])
   221→
   222→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   223→    if (e.target.files && e.target.files[0]) {
   224→      setSelectedFile(e.target.files[0])
   225→    }
   226→  }
   227→
   228→  const openUploadDialog = () => {
   229→    setSelectedFile(null); setUploadResult(null); setUploadProgress(0); setUploadDialogOpen(true)
   230→  }
   231→
   232→  const renderDept = (dept: Department, depth = 0) => {
   233→    const children = getChildren(dept.id)
   234→    const hasChildren = children.length > 0
   235→    const isExpanded = expandedIds.has(dept.id)
   236→    const isSelected = selectedDeptId === dept.id
   237→    return (
   238→      <div key={dept.id}>
   239→        <div className={`flex items-center gap-1 py-1.5 px-2 rounded-md cursor-pointer transition-colors ${isSelected ? 'bg-primary/10 text-primary font-medium' : 'hover:bg-muted/60'}`} style={{ paddingLeft: `${depth * 20 + 8}px` }} onClick={() => setSelectedDeptId(isSelected ? null : dept.id)}>
   240→          <button className="h-5 w-5 flex items-center justify-center flex-shrink-0 rounded hover:bg-muted" onClick={e => { e.stopPropagation(); toggleExpand(dept.id) }}>
   241→            {hasChildren ? (isExpanded ? <ChevronDown className="h-3.5 w-3.5" /> : <ChevronRight className="h-3.5 w-3.5" />) : <span className="h-3.5 w-3.5" />}
   242→          </button>
   243→          <Building2 className="h-4 w-4 flex-shrink-0 text-emerald-600" />
   244→          <span className="text-sm truncate flex-1">{dept.name}</span>
   245→          {getPosCount(dept.id) > 0 && <Badge variant="secondary" className="text-xs h-5 px-1.5">{getPosCount(dept.id)}</Badge>}
   246→          <div className="flex items-center gap-0.5 opacity-0 group-hover:opacity-100 hover:!opacity-100" style={{ opacity: undefined }}>
   247→            <button className="h-6 w-6 flex items-center justify-center rounded hover:bg-muted" onClick={e => { e.stopPropagation(); openCreateDept(dept.id) }} title="Добавить дочернее"><Plus className="h-3 w-3" /></button>
   248→            <button className="h-6 w-6 flex items-center justify-center rounded hover:bg-muted" onClick={e => { e.stopPropagation(); openEditDept(dept) }}><Pencil className="h-3 w-3" /></button>
   249→            <button className="h-6 w-6 flex items-center justify-center rounded hover:bg-destructive/10 hover:text-destructive" onClick={e => { e.stopPropagation(); setDeptToDelete(dept); setDeptDeleteOpen(true) }}><Trash2 className="h-3 w-3" /></button>
   250→          </div>
   251→        </div>
   252→        {hasChildren && isExpanded && children.map(c => renderDept(c, depth + 1))}
   253→      </div>
   254→    )
   255→  }
   256→
   257→  // Stats
   258→  const totalHeadcount = positions.reduce((sum, p) => sum + p.headcount, 0)
   259→  const uniqueDomains = [...new Set(positions.map(p => p.domain).filter(Boolean))]
   260→  const uniqueGrades = [...new Set(positions.map(p => p.grade).filter(Boolean))]
   261→
   262→  if (loading) return (
   263→    <div className="flex items-center justify-center py-12">
   264→      <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   265→    </div>
   266→  )
   267→
   268→  return (
   269→    <div className="space-y-6">
   270→      {/* Header */}
   271→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   272→        <div>
   273→          <h1 className="text-2xl font-bold flex items-center gap-2">
   274→            <Users className="h-6 w-6 text-emerald-600" /> 
   275→            Штатное расписание
   276→          </h1>
   277→          <p className="text-sm text-muted-foreground mt-1">Подразделения и должности организации</p>
   278→        </div>
   279→        <div className="flex gap-2 flex-wrap">
   280→          <Button variant="outline" onClick={openUploadDialog}>
   281→            <FileSpreadsheet className="h-4 w-4 mr-1.5" /> Загрузить файл
   282→          </Button>
   283→          <Button variant="outline" onClick={() => setBulkDialogOpen(true)}>
   284→            <Upload className="h-4 w-4 mr-1.5" /> Вставить текст
   285→          </Button>
   286→          <Button onClick={() => openCreateDept()} className="bg-emerald-600 hover:bg-emerald-700">
   287→            <Plus className="h-4 w-4 mr-1.5" /> Подразделение
   288→          </Button>
   289→          <Button variant="outline" onClick={() => openCreatePos()}>
   290→            <Plus className="h-4 w-4 mr-1.5" /> Должность
   291→          </Button>
   292→        </div>
   293→      </div>
   294→
   295→      {/* Stats */}
   296→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   297→        <Card className="bg-gradient-to-br from-emerald-50 to-emerald-100/50 border-emerald-200/50">
   298→          <CardContent className="p-3">
   299→            <p className="text-xs text-emerald-700 font-medium">Подразделения</p>
   300→            <p className="text-2xl font-bold text-emerald-800">{departments.length}</p>
   301→          </CardContent>
   302→        </Card>
   303→        <Card className="bg-gradient-to-br from-teal-50 to-teal-100/50 border-teal-200/50">
   304→          <CardContent className="p-3">
   305→            <p className="text-xs text-teal-700 font-medium">Должности</p>
   306→            <p className="text-2xl font-bold text-teal-800">{positions.length}</p>
   307→          </CardContent>
   308→        </Card>
   309→        <Card className="bg-gradient-to-br from-cyan-50 to-cyan-100/50 border-cyan-200/50">
   310→          <CardContent className="p-3">
   311→            <p className="text-xs text-cyan-700 font-medium">Штатных единиц</p>
   312→            <p className="text-2xl font-bold text-cyan-800">{totalHeadcount}</p>
   313→          </CardContent>
   314→        </Card>
   315→        <Card className="bg-gradient-to-br from-sky-50 to-sky-100/50 border-sky-200/50">
   316→          <CardContent className="p-3">
   317→            <p className="text-xs text-sky-700 font-medium">Доменов</p>
   318→            <p className="text-2xl font-bold text-sky-800">{uniqueDomains.length}</p>
   319→          </CardContent>
   320→        </Card>
   321→      </div>
   322→
   323→      {/* Main content */}
   324→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   325→        {/* Department tree */}
   326→        <Card className="lg:col-span-1">
   327→          <CardHeader className="pb-2">
   328→            <CardTitle className="text-base flex items-center gap-2">
   329→              <Building2 className="h-4 w-4 text-emerald-600" /> Подразделения
   330→            </CardTitle>
   331→          </CardHeader>
   332→          <CardContent className="max-h-[600px] overflow-y-auto">
   333→            {rootDepts.length === 0 ? (
   334→              <div className="text-center py-8 text-muted-foreground">
   335→                <Building2 className="h-8 w-8 mx-auto mb-2 opacity-40" />
   336→                <p className="text-sm">Нет подразделений</p>
   337→                <Button size="sm" className="mt-2" onClick={() => openCreateDept()}>Создать</Button>
   338→              </div>
   339→            ) : rootDepts.map(d => renderDept(d))}
   340→          </CardContent>
   341→        </Card>
   342→
   343→        {/* Positions table */}
   344→        <Card className="lg:col-span-2">
   345→          <CardHeader className="pb-2">
   346→            <div className="flex items-center justify-between">
   347→              <CardTitle className="text-base flex items-center gap-2">
   348→                <Users className="h-4 w-4 text-teal-600" /> Должности
   349→                {selectedDeptId && <Badge variant="outline" className="ml-1 text-emerald-700 border-emerald-300">{departments.find(d => d.id === selectedDeptId)?.name}</Badge>}
   350→              </CardTitle>
   351→            </div>
   352→          </CardHeader>
   353→          <CardContent className="p-0">
   354→            <div className="px-4 pb-2">
   355→              <div className="relative">
   356→                <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   357→                <Input placeholder="Поиск по названию, коду, грейду..." value={searchQuery} onChange={e => setSearchQuery(e.target.value)} className="pl-8" />
   358→              </div>
   359→            </div>
   360→            {filteredPositions.length === 0 ? (
   361→              <div className="text-center py-8 text-muted-foreground">
   362→                <Users className="h-8 w-8 mx-auto mb-2 opacity-40" />
   363→                <p className="text-sm">Нет должностей</p>
   364→              </div>
   365→            ) : (
   366→              <Table>
   367→                <TableHeader>
   368→                  <TableRow>
   369→                    <TableHead className="w-24">Код</TableHead>
